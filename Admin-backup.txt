
insert into storage.buckets (id, name, public)
values ('product-images', 'product-images', true)
on conflict (id) do update set public = true;

create policy "Anyone can view perfume images"
on storage.objects
for select
to anon, authenticated
using (bucket_id = 'product-images');

create policy "Admin can upload perfume images"
on storage.objects
for insert
to authenticated
with check (
  bucket_id = 'product-images'
  and auth.uid() = '1ebf9380-7d3b-4cca-a1c2-d4c1cb20290e'::uuid
);

create policy "Admin can update perfume images"
on storage.objects
for update
to authenticated
using (
  bucket_id = 'product-images'
  and auth.uid() = '1ebf9380-7d3b-4cca-a1c2-d4c1cb20290e'::uuid
)
with check (
  bucket_id = 'product-images'
  and auth.uid() = '1ebf9380-7d3b-4cca-a1c2-d4c1cb20290e'::uuid
);

create policy "Admin can delete perfume images"
on storage.objects
for delete
to authenticated
using (
  bucket_id = 'product-images'
  and auth.uid() = '1ebf9380-7d3b-4cca-a1c2-d4c1cb20290e'::uuid
);
